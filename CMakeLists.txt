cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/package.json" slang_package_json)
string(JSON SLANG_VERSION GET "${slang_package_json}" version)
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" slang_numeric_version "${SLANG_VERSION}")
project(Slang VERSION ${slang_numeric_version} LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Ship the Windows runtime in the executable; students need no VC redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# Pinned source dependency; only developers need the download. The distributed
# executable contains SDL, including its PNG loader, and works offline.
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
set(sdl_archive "https://github.com/libsdl-org/SDL/releases/download/release-3.4.18/SDL3-3.4.18.zip")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/build/downloads/SDL3-3.4.18.zip")
    set(sdl_archive "${CMAKE_CURRENT_SOURCE_DIR}/build/downloads/SDL3-3.4.18.zip")
endif()
FetchContent_Declare(SDL3 URL "${sdl_archive}"
    URL_HASH SHA256=9cd42377704398796071b8597cd7e21da254a43bad98c4199739647adc13fa6f
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(SDL3)

# Font data is built in: no operating-system font or extra file is required.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native/assets/NotoSans.ttf" slang_font_hex HEX)
string(REPEAT "[0-9a-f]" 64 slang_font_line_pattern)
string(REGEX REPLACE "(${slang_font_line_pattern})" "\\1\n" slang_font_hex "${slang_font_hex}")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," slang_font_bytes "${slang_font_hex}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/font_data.inc" CONTENT "${slang_font_bytes}" @ONLY)
include(native/aliases.cmake)

add_library(slang_core STATIC
    native/src/lexer.cpp
    native/src/locale.cpp
    native/src/ast.cpp
    native/src/parser.cpp
    native/src/number.cpp
    native/src/library.cpp
    native/src/font.cpp
    native/src/window.cpp
    native/src/graphics.cpp
    native/src/runtime.cpp)
target_include_directories(slang_core PUBLIC native/include)
target_include_directories(slang_core PRIVATE native/vendor "${CMAKE_CURRENT_BINARY_DIR}/generated")
# Upstream single-header implementation has its own warning conventions.
if(MSVC)
    set_source_files_properties(native/src/font.cpp PROPERTIES COMPILE_OPTIONS /w)
else()
    set_source_files_properties(native/src/font.cpp PROPERTIES COMPILE_OPTIONS -w)
endif()
target_link_libraries(slang_core PUBLIC SDL3::SDL3-static)
if(MSVC)
    target_compile_options(slang_core PRIVATE /W4 /utf-8 /permissive-)
else()
    target_compile_options(slang_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(slang native/src/main.cpp native/src/ide.cpp)
target_link_libraries(slang PRIVATE slang_core)
target_compile_definitions(slang PRIVATE SLANG_VERSION="${SLANG_VERSION}")
if(MSVC)
    target_compile_options(slang PRIVATE /W4 /utf-8 /permissive-)
    # The AST evaluator has larger frames than the future bytecode VM. Reserve
    # a consistent stack so its explicit recursion guard fires before overflow.
    target_link_options(slang PRIVATE /STACK:8388608)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(slang_tests native/tests/frontend_tests.cpp)
    target_link_libraries(slang_tests PRIVATE slang_core)
    if(MSVC)
        target_compile_options(slang_tests PRIVATE /W4 /utf-8 /permissive-)
    endif()
    add_test(NAME native_frontend COMMAND slang_tests)
    add_executable(slang_runtime_tests native/tests/runtime_tests.cpp)
    target_link_libraries(slang_runtime_tests PRIVATE slang_core)
    if(MSVC)
        target_compile_options(slang_runtime_tests PRIVATE /W4 /utf-8 /permissive-)
        target_link_options(slang_runtime_tests PRIVATE /STACK:8388608)
    endif()
    add_test(NAME native_runtime COMMAND slang_runtime_tests)
    add_executable(slang_module_tests native/tests/module_tests.cpp)
    target_link_libraries(slang_module_tests PRIVATE slang_core)
    if(MSVC)
        target_compile_options(slang_module_tests PRIVATE /W4 /utf-8 /permissive-)
        target_link_options(slang_module_tests PRIVATE /STACK:8388608)
    endif()
    add_test(NAME native_modules COMMAND slang_module_tests "${CMAKE_CURRENT_BINARY_DIR}/module-tests")
    add_executable(slang_library_tests native/tests/library_tests.cpp)
    target_link_libraries(slang_library_tests PRIVATE slang_core)
    if(MSVC)
        target_compile_options(slang_library_tests PRIVATE /W4 /utf-8 /permissive-)
        target_link_options(slang_library_tests PRIVATE /STACK:8388608)
    endif()
    add_test(NAME native_libraries COMMAND slang_library_tests "${CMAKE_CURRENT_BINARY_DIR}/library-tests")
    set_tests_properties(native_libraries PROPERTIES TIMEOUT 30)
    add_executable(slang_graphics_tests native/tests/graphics_tests.cpp)
    target_link_libraries(slang_graphics_tests PRIVATE slang_core)
    if(MSVC)
        target_compile_options(slang_graphics_tests PRIVATE /W4 /utf-8 /permissive-)
        target_link_options(slang_graphics_tests PRIVATE /STACK:8388608)
    endif()
    add_test(NAME native_graphics COMMAND slang_graphics_tests "${CMAKE_CURRENT_BINARY_DIR}/graphics-tests")
    set_tests_properties(native_graphics PROPERTIES ENVIRONMENT "SDL_VIDEO_DRIVER=dummy;SDL_RENDER_DRIVER=software" TIMEOUT 30)
    add_executable(slang_window_tests native/tests/window_tests.cpp)
    target_link_libraries(slang_window_tests PRIVATE slang_core)
    if(MSVC)
        target_compile_options(slang_window_tests PRIVATE /W4 /utf-8 /permissive-)
        target_link_options(slang_window_tests PRIVATE /STACK:8388608)
    endif()
    add_test(NAME native_windows COMMAND slang_window_tests "${CMAKE_CURRENT_BINARY_DIR}/window-tests")
    set_tests_properties(native_windows PROPERTIES ENVIRONMENT "SDL_VIDEO_DRIVER=dummy;SDL_RENDER_DRIVER=software" TIMEOUT 30)
    add_test(NAME english_example COMMAND slang check "${CMAKE_CURRENT_SOURCE_DIR}/examples/english.slang")
    add_test(NAME norwegian_example COMMAND slang check "${CMAKE_CURRENT_SOURCE_DIR}/examples/norsk.slang")
    add_test(NAME run_english_example COMMAND slang run "${CMAKE_CURRENT_SOURCE_DIR}/examples/english.slang")
    add_test(NAME run_norwegian_example COMMAND slang run "${CMAKE_CURRENT_SOURCE_DIR}/examples/norsk.slang")
    add_test(NAME run_first_program COMMAND slang run "${CMAKE_CURRENT_SOURCE_DIR}/examples/first-program.slang")
    add_test(NAME run_module_example COMMAND slang run "${CMAKE_CURRENT_SOURCE_DIR}/examples/modules/main.slang")
    foreach(example dice word-game measurements norsk)
        add_test(NAME standard_library_${example} COMMAND slang run "${CMAKE_CURRENT_SOURCE_DIR}/examples/standard-library/${example}.slang")
    endforeach()
    # Keep the notebook's intentional file writes inside the build directory.
    configure_file(examples/standard-library/notebook.slang library-examples/notebook.slang COPYONLY)
    add_test(NAME standard_library_notebook COMMAND slang run "${CMAKE_CURRENT_BINARY_DIR}/library-examples/notebook.slang")
    add_test(NAME native_cli COMMAND ${CMAKE_COMMAND}
        -DSLANG_EXE=$<TARGET_FILE:slang>
        -DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/cli-tests
        -P ${CMAKE_CURRENT_SOURCE_DIR}/native/tests/cli_tests.cmake)
endif()

install(TARGETS slang RUNTIME DESTINATION bin)

install(FILES examples/graphics/LICENSE DESTINATION examples/graphics)
install(FILES native/assets/OFL.txt DESTINATION licenses/NotoSans)
install(FILES native/vendor/stb/LICENSE DESTINATION licenses/stb_truetype)
install(FILES "${sdl3_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses/SDL3)

install(FILES "${sdl3_SOURCE_DIR}/src/video/yuv2rgb/LICENSE" DESTINATION licenses/SDL3 RENAME yuv2rgb.txt)
# These upstream single-file libraries carry their notices in the source.
install(FILES "${sdl3_SOURCE_DIR}/src/video/stb_image.h" "${sdl3_SOURCE_DIR}/src/video/miniz.h" DESTINATION licenses/SDL3)
install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md")
install(FILES README.md LICENSE THIRD-PARTY-NOTICES.md DESTINATION .)
install(FILES packaging/START-HERE.txt DESTINATION .)
if(WIN32)
    install(FILES packaging/SLANG.cmd DESTINATION .)
endif()
set(CPACK_PACKAGE_NAME "slang")
set(CPACK_PACKAGE_VERSION ${SLANG_VERSION})
if(WIN32)
    set(CPACK_GENERATOR ZIP)
else()
    set(CPACK_GENERATOR TGZ)
endif()
set(CPACK_PACKAGE_FILE_NAME "slang-${SLANG_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
